## Context

R is the most widely used statistical programming language in academic
research. Its `haven` package reads/writes SPSS files natively. By generating
R scripts and executing them via `Rscript`, Carina can leverage the full R
ecosystem without needing PSPP or Jamovi. This is the most powerful path
but requires R to be installed on the host.

---

## Goal

Build an R script generation and execution pipeline:

```
User: "Run multiple regression predicting Y from X1, X2, X3 in mydata.sav"
  -> Carina generates analysis.R
  -> Carina runs: Rscript analysis.R
  -> Carina parses output
  -> Carina returns APA-formatted results
```

### Step 1: R Script Generator

Create `02-backends/core.carinaai.uk/src/tools/research/r-generator.ts`:

```typescript
export function generateRScript(params: {
  dataFile: string;
  analysisType: 'descriptive' | 'ttest' | 'anova' | 'correlation' |
                'regression' | 'chi-square' | 'factor' | 'reliability';
  variables: string[];
  groupBy?: string;
  dependentVar?: string;
  independentVars?: string[];
  apaFormat?: boolean;
}): string {
  const lines = [
    `# R analysis script generated by Carina`,
    `# Analysis: ${params.analysisType}`,
    `# Generated: ${new Date().toISOString()}`,
    ``,
    `# Load libraries`,
    `library(haven)`,
    `library(dplyr)`,
    `library(tidyr)`,
  ];

  if (params.apaFormat) {
    lines.push(`library(apaTables)`);
  }

  lines.push(
    ``,
    `# Load data`,
    `df <- read_sav("${params.dataFile}")`,
    `cat("Data loaded:", nrow(df), "rows,", ncol(df), "columns\\n")`,
    ``,
  );

  switch (params.analysisType) {
    case 'descriptive':
      lines.push(
        `# Descriptive statistics`,
        `vars <- c(${quoteList(params.variables)})`,
        `desc <- df %>%`,
        `  select(all_of(vars)) %>%`,
        `  psych::describe() %>%`,
        `  as.data.frame()`,
        `print(desc)`,
      );
      break;

    case 'ttest':
      lines.push(
        `# Independent samples t-test`,
        `for (var in c(${quoteList(params.variables)})) {`,
        `  cat("\\n--- T-test for", var, "---\\n")`,
        `  formula <- as.formula(paste(var, "~", "${params.groupBy}"))`,
        `  result <- t.test(formula, data = df)`,
        `  print(result)`,
        `  # Effect size (Cohen's d)`,
        `  group_means <- tapply(df[[var]], df[["${params.groupBy}"]], mean, na.rm = TRUE)`,
        `  group_sds <- tapply(df[[var]], df[["${params.groupBy}"]], sd, na.rm = TRUE)`,
        `  pooled_sd <- sqrt(mean(group_sds^2))`,
        `  d <- abs(diff(group_means)) / pooled_sd`,
        `  cat("Cohen's d:", round(d, 3), "\\n")`,
        `}`,
      );
      break;

    case 'anova':
      lines.push(
        `# One-way ANOVA`,
        `for (var in c(${quoteList(params.variables)})) {`,
        `  cat("\\n--- ANOVA for", var, "---\\n")`,
        `  formula <- as.formula(paste(var, "~", "${params.groupBy}"))`,
        `  model <- aov(formula, data = df)`,
        `  print(summary(model))`,
        `  # Eta-squared`,
        `  eta_sq <- effectsize::eta_squared(model)`,
        `  print(eta_sq)`,
        `  # Tukey HSD post-hoc`,
        `  cat("\\nTukey HSD post-hoc:\\n")`,
        `  print(TukeyHSD(model))`,
        `}`,
      );
      break;

    case 'regression':
      lines.push(
        `# Multiple linear regression`,
        `formula <- as.formula(paste("${params.dependentVar} ~",`,
        `  paste(c(${quoteList(params.independentVars || params.variables)}), collapse = " + ")))`,
        `model <- lm(formula, data = df)`,
        `cat("\\n--- Regression Results ---\\n")`,
        `print(summary(model))`,
        ``,
        `# Standardized coefficients`,
        `cat("\\nStandardised coefficients (beta):\\n")`,
        `df_std <- df %>% mutate(across(where(is.numeric), scale))`,
        `model_std <- lm(formula, data = df_std)`,
        `print(coef(model_std))`,
        ``,
        `# Diagnostics`,
        `cat("\\n--- Assumption Checks ---\\n")`,
        `# VIF for multicollinearity`,
        `if (length(coef(model)) > 2) {`,
        `  cat("VIF values:\\n")`,
        `  print(car::vif(model))`,
        `}`,
      );
      break;

    case 'correlation':
      lines.push(
        `# Correlation matrix`,
        `cor_matrix <- df %>%`,
        `  select(c(${quoteList(params.variables)})) %>%`,
        `  cor(use = "pairwise.complete.obs")`,
        `cat("Correlation matrix:\\n")`,
        `print(round(cor_matrix, 3))`,
        ``,
        `# Significance tests`,
        `cat("\\nCorrelation with p-values:\\n")`,
        `psych::corr.test(df[, c(${quoteList(params.variables)})], use = "pairwise")`,
      );
      break;

    case 'chi-square':
      lines.push(
        `# Chi-square test of independence`,
        `if (length(c(${quoteList(params.variables)})) >= 2) {`,
        `  var1 <- ${quoteList(params.variables)}[1]`,
        `  var2 <- ${quoteList(params.variables)}[2]`,
        `  cat("\\n--- Chi-square:", var1, "x", var2, "---\\n")`,
        `  tbl <- table(df[[var1]], df[[var2]])`,
        `  print(tbl)`,
        `  result <- chisq.test(tbl)`,
        `  print(result)`,
        `  # Cramer's V`,
        `  cramer_v <- sqrt(result$statistic / (sum(tbl) * (min(dim(tbl)) - 1)))`,
        `  cat("Cramer's V:", round(cramer_v, 3), "\\n")`,
        `}`,
      );
      break;

    case 'factor':
      lines.push(
        `# Exploratory Factor Analysis`,
        `library(psych)`,
        `fa_result <- fa(df[, c(${quoteList(params.variables)})], nfactors = 1, rotate = "varimax")`,
        `print(fa_result$loadings, cutoff = 0.3)`,
        `cat("\\nEigenvalues:\\n")`,
        `print(fa_result$values)`,
        `cat("\\nVariance explained:", round(sum(fa_result$values[1]) / length(fa_result$values) * 100, 1), "%\\n")`,
      );
      break;

    case 'reliability':
      lines.push(
        `# Reliability analysis (Cronbach's alpha)`,
        `library(psych)`,
        `alpha_result <- alpha(df[, c(${quoteList(params.variables)})])`,
        `cat("Cronbach's alpha:", round(alpha_result$total$raw_alpha, 3), "\\n")`,
        `cat("\\nItem-total statistics:\\n")`,
        `print(alpha_result$item.stats)`,
      );
      break;
  }

  return lines.join('\n');
}

function quoteList(items: string[]): string {
  return items.map(v => `"${v}"`).join(', ');
}
```

### Step 2: R Executor Tool

Create `02-backends/core.carinaai.uk/src/tools/research/r-executor.ts`:

```typescript
import { execSync } from 'node:child_process';
import { writeFileSync, unlinkSync } from 'node:fs';
import { randomUUID } from 'node:crypto';
import { generateRScript } from './r-generator';

interface RAvailability {
  installed: boolean;
  version?: string;
  packages?: { haven: boolean; dplyr: boolean; psych: boolean };
}

function checkR(): RAvailability {
  try {
    const version = execSync('Rscript --version', { encoding: 'utf-8', timeout: 5000 }).trim();

    // Check required packages
    const pkgCheck = (pkg: string): boolean => {
      try {
        execSync(`Rscript -e 'library(${pkg})' 2>&1`, { timeout: 5000 });
        return true;
      } catch { return false; }
    };

    return {
      installed: true,
      version,
      packages: {
        haven: pkgCheck('haven'),
        dplyr: pkgCheck('dplyr'),
        psych: pkgCheck('psych'),
      },
    };
  } catch {
    return { installed: false };
  }
}

export async function executeRTool(params: {
  dataFile: string;
  analysisType: string;
  variables: string[];
  groupBy?: string;
  dependentVar?: string;
  independentVars?: string[];
  outputFormat?: 'text' | 'json' | 'apa';
}): Promise<ToolResult> {
  // 1. Check R availability
  const r = checkR();
  if (!r.installed) {
    return {
      content: [{
        type: 'text',
        text: `R is not installed. Install with: sudo apt install r-base\n` +
          `Then install required packages:\n` +
          `Rscript -e 'install.packages(c("haven","dplyr","psych","broom","car"))'\n\n` +
          `Falling back to Python analysis. Use stats_describe or stats_compare_groups instead.`
      }],
      isError: true,
    };
  }

  // 2. Check required packages
  const missingPkgs = Object.entries(r.packages || {})
    .filter(([, has]) => !has)
    .map(([pkg]) => pkg);
  if (missingPkgs.length > 0) {
    return {
      content: [{
        type: 'text',
        text: `R packages missing: ${missingPkgs.join(', ')}\n` +
          `Install: Rscript -e 'install.packages(c("${missingPkgs.join('","')}"))'`
      }],
      isError: true,
    };
  }

  // 3. Generate R script
  const script = generateRScript({
    dataFile: params.dataFile,
    analysisType: params.analysisType as any,
    variables: params.variables,
    groupBy: params.groupBy,
    dependentVar: params.dependentVar,
    independentVars: params.independentVars,
    apaFormat: params.outputFormat === 'apa',
  });

  const scriptFile = `/tmp/carina-r-${randomUUID().slice(0, 8)}.R`;

  try {
    writeFileSync(scriptFile, script);

    // 4. Execute R script
    const output = execSync(
      `Rscript "${scriptFile}" 2>&1`,
      { encoding: 'utf-8', timeout: 120000, maxBuffer: 10 * 1024 * 1024 }
    );

    return {
      content: [{
        type: 'text',
        text: `## R Analysis Results\n\n` +
          `**Analysis:** ${params.analysisType}\n` +
          `**Data:** ${params.dataFile}\n` +
          `**R version:** ${r.version}\n\n` +
          `\`\`\`\n${output.slice(0, 8000)}${output.length > 8000 ? '\n... (truncated)' : ''}\n\`\`\``
      }],
    };
  } catch (err: any) {
    return {
      content: [{
        type: 'text',
        text: `R script execution failed: ${err.message}\nScript saved at: ${scriptFile}`
      }],
      isError: true,
    };
  } finally {
    try { unlinkSync(scriptFile); } catch {}
  }
}
```

### Step 3: Tool Registration

```typescript
{
  name: 'r_analyze',
  description: 'Run statistical analysis using R. Generates R script, executes ' +
    'via Rscript, returns formatted output. Requires R and packages: haven, ' +
    'dplyr, psych, broom, car. Supports: descriptive, t-test, ANOVA, correlation, ' +
    'regression, chi-square, factor analysis, reliability.',
  parameters: {
    type: 'object',
    properties: {
      dataFile: { type: 'string', description: 'Path to .sav or .csv file' },
      analysisType: {
        type: 'string',
        enum: ['descriptive', 'ttest', 'anova', 'correlation', 'regression',
               'chi-square', 'factor', 'reliability'],
      },
      variables: { type: 'array', items: { type: 'string' } },
      groupBy: { type: 'string' },
      dependentVar: { type: 'string' },
      independentVars: { type: 'array', items: { type: 'string' } },
      outputFormat: { type: 'string', enum: ['text', 'json', 'apa'], default: 'text' },
    },
    required: ['dataFile', 'analysisType', 'variables'],
  },
  requires: {
    commands: ['Rscript'],
    rPackages: ['haven', 'dplyr', 'psych'],
  },
  handler: executeRTool,
}
```
