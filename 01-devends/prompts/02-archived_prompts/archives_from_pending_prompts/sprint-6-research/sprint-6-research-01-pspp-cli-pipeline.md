## Context

PSPP is a free, open-source clone of SPSS. It uses the same syntax (.sps files)
and reads/writes the same data formats (.sav, .spv). Because it's CLI-driven,
Carina can automate the full workflow: write syntax, execute PSPP, parse output,
format results. No API key, no cloud dependency, zero cost.

---

## Goal

Build a complete PSPP analysis pipeline:

```
User: "Compare group A and B on variable X in mydata.sav"
  -> Carina writes analysis.sps
  -> Carina runs: pspp mydata.sav analysis.sps
  -> Carina parses output.txt
  -> Carina returns APA-formatted results
```

```typescript
import { execSync } from 'node:child_process';

export interface PsppAvailability {
  installed: boolean;
  version?: string;
  path?: string;
  error?: string;
}

export function checkPspp(): PsppAvailability {
  try {
    const version = execSync('pspp --version', { encoding: 'utf-8', timeout: 5000 }).trim();
    const path = execSync('which pspp', { encoding: 'utf-8', timeout: 3000 }).trim();
    return { installed: true, version, path };
  } catch (err) {
    return { installed: false, error: 'PSPP not installed. Run: sudo apt install pspp' };
  }
}
```

### Step 2: Syntax Generator

Create `02-backends/core.carinaai.uk/src/tools/research/pspp-syntax.ts`:

```typescript
export interface SyntaxRequest {
  dataFile: string; // path to .sav or .csv
  analysisType: 'descriptive' | 'ttest' | 'anova' | 'chi-square' |
                'correlation' | 'regression' | 'nonparametric' | 'factor';
  variables?: string[]; // variables to analyse
  groupBy?: string; // grouping variable for t-test/ANOVA
  dependentVar?: string; // DV for regression
  independentVars?: string[]; // IVs for regression
  options?: {
    posthoc?: 'bonferroni' | 'tukey' | 'none';
    normality?: boolean;
    effectSize?: boolean;
    missingListwise?: boolean;
  };
}

export function generateSyntax(req: SyntaxRequest): string {
  const lines: string[] = [
    `* PSPP syntax generated by Carina`,
    `* Analysis: ${req.analysisType}`,
    `* Generated at: ${new Date().toISOString()}`,
    ``,
    `GET FILE='${req.dataFile}'.`,
    `DATASET NAME DataSet1 WINDOW=FRONT.`,
    ``,
  ];

  switch (req.analysisType) {
    case 'descriptive':
      lines.push(
        `DESCRIPTIVES VARIABLES=${req.variables?.join(' ')}`,
        `  /STATISTICS=MEAN STDDEV MIN MAX SEMEAN KURTOSIS SKEWNESS.`,
        ``,
        `FREQUENCIES VARIABLES=${req.variables?.join(' ')}`,
        `  /STATISTICS=MEDIAN MODE`,
        `  /HISTOGRAM NORMAL`,
        `  /ORDER=ANALYSIS.`,
      );
      break;

    case 'ttest':
      lines.push(
        `T-TEST GROUPS=${req.groupBy}(1,2)`,
        `  /MISSING=ANALYSIS`,
        `  /VARIABLES=${req.variables?.join(' ')}`,
        `  /ES DISPLAY(TRUE).`,
      );
      if (req.options?.effectSize) {
        lines.push(`  /CRITERIA=CI(.95).`);
      }
      break;

    case 'anova':
      lines.push(
        `ONEWAY ${req.variables?.join(' ')} BY ${req.groupBy}`,
        `  /MISSING ANALYSIS.`,
      );
      if (req.options?.posthoc && req.options.posthoc !== 'none') {
        const test = req.options.posthoc === 'bonferroni' ? 'BONFERRONI' : 'TUKEY';
        lines.push(`  /POSTHOC=${test} ALPHA(0.05).`);
      }
      if (req.options?.effectSize) {
        lines.push(`  /STATISTICS EFFECTS.`);
      }
      break;

    case 'chi-square':
      lines.push(
        `CROSSTABS`,
        `  /TABLES=${req.variables?.[0]} BY ${req.variables?.[1]}`,
        `  /FORMAT=AVALUE TABLES`,
        `  /STATISTICS=CHISQ PHI`,
        `  /CELLS=COUNT EXPECTED ROW COLUMN TOTAL`,
        `  /COUNT ROUND CELL.`,
      );
      break;

    case 'correlation':
      lines.push(
        `CORRELATIONS`,
        `  /VARIABLES=${req.variables?.join(' ')}`,
        `  /PRINT=TWOTAIL NOSIG`,
        `  /MISSING=PAIRWISE.`,
      );
      break;

    case 'regression':
      lines.push(
        `REGRESSION`,
        `  /MISSING LISTWISE`,
        `  /STATISTICS COEFF OUTS R ANOVA`,
        `  /CRITERIA=PIN(.05) POUT(.10)`,
        `  /NOORIGIN`,
        `  /DEPENDENT ${req.dependentVar}`,
        `  /METHOD=ENTER ${req.independentVars?.join(' ')}.`,
      );
      break;

    case 'nonparametric':
      lines.push(
        `NPAR TESTS`,
        `  /M-W= ${req.variables?.join(' ')} BY ${req.groupBy}(1 2)`,
        `  /MISSING ANALYSIS.`,
      );
      break;

    case 'factor':
      lines.push(
        `FACTOR`,
        `  /VARIABLES ${req.variables?.join(' ')}`,
        `  /MISSING LISTWISE`,
        `  /ANALYSIS ${req.variables?.join(' ')}`,
        `  /PRINT INITIAL EXTRACTION ROTATION`,
        `  /CRITERIA MINEIGEN(1) ITERATE(25)`,
        `  /EXTRACTION PC`,
        `  /ROTATION VARIMAX.`,
      );
      break;
  }

  return lines.join('\n');
}
```

### Step 3: PSPP Executor Tool

Create `02-backends/core.carinaai.uk/src/tools/research/pspp-executor.ts`:

```typescript
import { execSync } from 'node:child_process';
import { writeFileSync, readFileSync, unlinkSync } from 'node:fs';
import { randomUUID } from 'node:crypto';
import { checkPspp } from './pspp-check';
import { generateSyntax, type SyntaxRequest } from './pspp-syntax';

interface PsppResult {
  success: boolean;
  output?: string;
  error?: string;
  syntaxFile?: string;
  outputFile?: string;
  parsedResults?: ParsedPsppOutput;
}

interface ParsedPsppOutput {
  tables: PsppTable[];
  notes: string[];
  warnings: string[];
  errors: string[];
}

interface PsppTable {
  title: string;
  rows: string[][];
}

export async function executePsppTool(
  params: SyntaxRequest & { outputFormat?: 'text' | 'json' | 'apa' }
): Promise<ToolResult> {
  // 1. Check PSPP availability
  const pspp = checkPspp();
  if (!pspp.installed) {
    return {
      content: [{
        type: 'text',
        text: `PSPP is not installed. Install it with: sudo apt install pspp\nFalling back to Python analysis. Use the stats_describe or stats_compare_groups tools instead.`
      }],
      isError: true,
    };
  }

  // 2. Generate syntax
  const syntax = generateSyntax(params);
  const syntaxFile = `/tmp/carina-pspp-${randomUUID().slice(0, 8)}.sps`;
  const outputFile = `/tmp/carina-pspp-${randomUUID().slice(0, 8)}.txt`;

  try {
    // 3. Write syntax file
    writeFileSync(syntaxFile, syntax);

    // 4. Execute PSPP
    const result = execSync(
      `pspp "${params.dataFile}" "${syntaxFile}" 2>&1`,
      { encoding: 'utf-8', timeout: 60000, maxBuffer: 10 * 1024 * 1024 }
    );

    // 5. Get output (PSPP writes to pspp.txt by default, or capture stdout)
    let output: string;
    try {
      // PSPP output goes to pspp.txt in the current directory
      output = readFileSync('pspp.txt', 'utf-8');
    } catch {
      output = result; // fall back to stdout
    }

    // 6. Parse output
    const parsed = parsePsppOutput(output);

    // 7. Format based on requested format
    if (params.outputFormat === 'apa') {
      const apaFormatted = formatAsAPA(parsed, params);
      return {
        content: [{ type: 'text', text: apaFormatted }],
      };
    }

    if (params.outputFormat === 'json') {
      return {
        content: [{ type: 'text', text: JSON.stringify(parsed, null, 2) }],
      };
    }

    // Default: return raw output with summary
    return {
      content: [{
        type: 'text',
        text: `PSPP analysis complete.\n\nTables found: ${parsed.tables.length}\nWarnings: ${parsed.warnings.length}\nErrors: ${parsed.errors.length}\n\n--- PSPP Output ---\n${output.slice(0, 5000)}${output.length > 5000 ? '\n... (truncated)' : ''}`
      }],
    };
  } catch (err: any) {
    return {
      content: [{
        type: 'text',
        text: `PSPP execution failed: ${err.message}\nSyntax file saved at: ${syntaxFile}`
      }],
      isError: true,
    };
  } finally {
    // Cleanup temp files
    try { unlinkSync(syntaxFile); } catch {}
  }
}

/** Parse PSPP text output into structured tables */
function parsePsppOutput(output: string): ParsedPsppOutput {
  const tables: PsppTable[] = [];
  const notes: string[] = [];
  const warnings: string[] = [];
  const errors: string[] = [];

  const lines = output.split('\n');

  // Simple state machine to extract tables
  let currentTable: PsppTable | null = null;
  let inTable = false;

  for (const line of lines) {
    // Detect notes/warnings/errors
    if (line.startsWith('Note:')) notes.push(line);
    if (line.startsWith('Warning')) warnings.push(line);
    if (line.startsWith('error:') || line.includes('Error:')) errors.push(line);

    // Detect table headers (PSPP uses separator lines with === or ---)
    if (line.match(/^[=\-]{10,}$/)) {
      if (inTable && currentTable) {
        tables.push(currentTable);
        currentTable = null;
      }
      inTable = !inTable;
      continue;
    }

    if (inTable) {
      if (!currentTable) {
        currentTable = { title: line.trim(), rows: [] };
      } else {
        const cells = line.trim().split(/\s{2,}/);
        if (cells.length > 1 && cells[0]) {
          currentTable.rows.push(cells);
        }
      }
    }
  }

  // Push last table if still open
  if (currentTable && currentTable.rows.length > 0) {
    tables.push(currentTable);
  }

  return { tables, notes, warnings, errors };
}

/** Format parsed output in APA 7th edition style */
function formatAsAPA(parsed: ParsedPsppOutput, params: SyntaxRequest): string {
  const lines: string[] = [
    `# Statistical Analysis Results`,
    ``,
    `**Analysis type:** ${params.analysisType}`,
    `**Data file:** ${params.dataFile}`,
    `**Variables:** ${params.variables?.join(', ') || 'all'}`,
    ``,
  ];

  // Extract key statistics from tables and format APA-style
  for (const table of parsed.tables) {
    lines.push(`### ${table.title}`);
    lines.push('');
    lines.push('| ' + table.rows[0]?.join(' | ') + ' |');
    lines.push('|' + table.rows[0]?.map(() => '---').join('|') + '|');
    for (const row of table.rows.slice(1)) {
      lines.push('| ' + row.join(' | ') + ' |');
    }
    lines.push('');
  }

  if (parsed.warnings.length > 0) {
    lines.push(`**Warnings:** ${parsed.warnings.join('; ')}`);
    lines.push('');
  }

  return lines.join('\n');
}
```

### Step 4: Tool Registration

Register the PSPP tool in Carina's tool registry:

```typescript
// In packages/packs/research-analysis/tools.ts
{
  name: 'pspp_analyze',
  description: 'Run statistical analysis using PSPP (free SPSS clone). ' +
    'Generates SPSS syntax, executes PSPP, parses output, returns APA-formatted results. ' +
    'Supports: descriptive, t-test, ANOVA, chi-square, correlation, regression, nonparametric, factor analysis.',
  parameters: {
    type: 'object',
    properties: {
      dataFile: { type: 'string', description: 'Path to .sav or .csv data file' },
      analysisType: {
        type: 'string',
        enum: ['descriptive', 'ttest', 'anova', 'chi-square', 'correlation', 'regression', 'nonparametric', 'factor'],
      },
      variables: { type: 'array', items: { type: 'string' }, description: 'Variables to analyse' },
      groupBy: { type: 'string', description: 'Grouping variable for t-test/ANOVA' },
      dependentVar: { type: 'string', description: 'Dependent variable for regression' },
      independentVars: { type: 'array', items: { type: 'string' }, description: 'Independent variables for regression' },
      options: {
        type: 'object',
        properties: {
          posthoc: { type: 'string', enum: ['bonferroni', 'tukey', 'none'] },
          normality: { type: 'boolean' },
          effectSize: { type: 'boolean' },
        },
      },
      outputFormat: { type: 'string', enum: ['text', 'json', 'apa'], default: 'apa' },
    },
    required: ['dataFile', 'analysisType'],
  },
  handler: executePsppTool,
}
```
