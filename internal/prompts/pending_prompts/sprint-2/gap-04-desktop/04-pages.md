## 6. Canvas Window

When user clicks "Open in Canvas" for an artifact, create a second Tauri window:

```typescript
// src/lib/canvas-window.ts

import { WebviewWindow } from '@tauri-apps/api/webviewWindow';

export async function openCanvasWindow(artifact: Artifact) {
  const window = new WebviewWindow(`canvas-${artifact.id}`, {
    url: `/canvas?id=${artifact.id}`,
    title: artifact.title || 'Canvas',
    width: 900,
    height: 700,
    decorations: true,
    resizable: true,
  });
  return window;
}
```
